# IHG World Cup Site — Plan 3: Sports-Tracker Frontend

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task.

**Goal:** Build the static HTML/CSS/TypeScript frontend that renders `site/data/matches.json` as a sports-tracker dashboard. The site shows IHG's 11 World Cup 2026 matches as a chronological list of tiles (timeline strip on top, hero card for the next match, then 11 tiles), with per-feeder team distributions as the primary scenario display for TBD matches and cross-product top-3 as supporting detail. Polls `matches.json` every 60s and diff-renders tiles whose signature changed.

**Architecture:** Vanilla TypeScript (no framework), bundled via `esbuild` to a single JS file. CSS hand-authored with design tokens. Country flags from a bundled SVG set. Output served as static files from `site/` (which GitHub Pages will serve in Plan 4). All rendering is template-literal-based for readability; tile state is keyed by match `id` and gated by `signature` for diff updates.

**Tech Stack:** TypeScript 5+, esbuild, vitest (for pure-logic unit tests), no UI framework. Static output: `site/index.html` (hand-authored), `site/assets/app.js` (bundled), `site/assets/styles.css` (hand-authored), `site/assets/flags/*.svg` (bundled subset).

**Reference spec:** [`docs/superpowers/specs/2026-05-08-world-cup-intelligence-site-design.md`](../specs/2026-05-08-world-cup-intelligence-site-design.md), Section 5 (frontend).

**Predecessor:** Plan 2, tagged `plan-2-complete`.

```typescript
import type { DataFreshness, TournamentPhase } from "../types.js";

export function formatProbability(p: number): string {
  return `${(p * 100).toFixed(1)}%`;
}

export function formatLocalKickoff(iso: string): string {
  // Parse the ISO string preserving its embedded timezone offset.
  const date = new Date(iso);
  const month = date.toLocaleString("en-US", { month: "short", timeZone: "UTC" });
  const day = date.toLocaleString("en-US", { day: "numeric", timeZone: "UTC" });
  // Use the embedded offset by subtracting the fact that UTC parsing shifts.
  // Simpler approach: format the time part by manipulating the ISO string.
  const m = iso.match(/T(\d{2}):(\d{2})/);
  if (!m) return `${month} ${day}`;
  const h24 = parseInt(m[1], 10);
  const mm = m[2];
  const ampm = h24 >= 12 ? "PM" : "AM";
  const h12 = ((h24 + 11) % 12) + 1;
  return `${month} ${day} · ${h12}:${mm} ${ampm}`;
}

const FRESH_THRESHOLD_MS: Record<TournamentPhase, number> = {
  pre_tournament: 60 * 60 * 1000,
  group_stage: 45 * 60 * 1000,
  round_of_32: 20 * 60 * 1000,
  round_of_16: 20 * 60 * 1000,
  quarter_finals: 20 * 60 * 1000,
  semi_finals: 20 * 60 * 1000,
  finals: 20 * 60 * 1000,
};

export function computeWallClockFreshness(
  generatedAtIso: string,
  serverFreshness: DataFreshness,
  phase: TournamentPhase,
  now: Date = new Date(),
): DataFreshness {
  if (serverFreshness === "unreachable") return "unreachable";
  const generatedAt = new Date(generatedAtIso);
  const ageMs = now.getTime() - generatedAt.getTime();
  const threshold = FRESH_THRESHOLD_MS[phase];
  if (ageMs > threshold) return "stale";
  return serverFreshness;
}

export function formatRelativeTime(generatedAtIso: string, now: Date = new Date()): string {
  const generatedAt = new Date(generatedAtIso);
  const ageSec = Math.floor((now.getTime() - generatedAt.getTime()) / 1000);
  if (ageSec < 60) return `${ageSec}s ago`;
  if (ageSec < 3600) return `${Math.floor(ageSec / 60)}m ago`;
  return `${Math.floor(ageSec / 3600)}h ago`;
}
```

```typescript
import type { MatchesFile } from "../types.js";

export function signaturesByMatchId(file: MatchesFile): Map<string, string> {
  const map = new Map<string, string>();
  for (const m of file.matches) map.set(m.id, m.signature);
  return map;
}

export function changedMatchIds(
  previous: MatchesFile | null,
  next: MatchesFile,
): string[] {
  if (previous === null) return next.matches.map((m) => m.id);
  const prev = signaturesByMatchId(previous);
  return next.matches
    .filter((m) => prev.get(m.id) !== m.signature)
    .map((m) => m.id);
}
```

```typescript
import type { MatchesFile } from "../types.js";

export type FetchResult =
  | { kind: "ok"; file: MatchesFile }
  | { kind: "unreachable"; error: string };

export async function fetchMatches(
  url: string,
  fetchImpl: typeof fetch = fetch,
): Promise<FetchResult> {
  try {
    const resp = await fetchImpl(url, { cache: "no-cache" });
    if (!resp.ok) {
      return { kind: "unreachable", error: `HTTP ${resp.status}` };
    }
    const file = (await resp.json()) as MatchesFile;
    return { kind: "ok", file };
  } catch (e) {
    return { kind: "unreachable", error: String(e) };
  }
}
```

```typescript
import type { MatchObject } from "../types.js";

export function renderTimeline(matches: MatchObject[]): string {
  if (matches.length === 0) return "";
  const sorted = [...matches].sort(
    (a, b) => new Date(a.kickoff_utc).getTime() - new Date(b.kickoff_utc).getTime(),
  );
  const tFirst = new Date(sorted[0].kickoff_utc).getTime();
  const tLast = new Date(sorted[sorted.length - 1].kickoff_utc).getTime();
  const span = Math.max(tLast - tFirst, 1);

  const markers = sorted.map((m) => {
    const t = new Date(m.kickoff_utc).getTime();
    const pct = ((t - tFirst) / span) * 100;
    const tierClass = `tier-${m.demand_tier}`;
    const stateClass = m.status === "tbd" ? "is-tbd" : "is-confirmed";
    const phaseLabel = phaseShort(m.phase);
    const dateLabel = monthDay(m.kickoff_utc);
    return `
      <a class="tl-marker ${stateClass} ${tierClass}" href="#match-${m.id}"
         style="left: ${pct.toFixed(2)}%"
         title="${m.host_city} · ${m.venue} · ${phaseLabel}">
        <span class="tl-dot"></span>
        <span class="tl-label">
          <span class="tl-date">${dateLabel}</span>
          <span class="tl-phase">${phaseLabel}</span>
          <span class="tl-city">${m.host_city}</span>
        </span>
      </a>
    `;
  }).join("");

  return `
    <div class="timeline">
      <div class="tl-rail"></div>
      ${markers}
    </div>
    <div class="timeline-legend">
      <span><span class="tl-dot is-confirmed tier-high"></span> high</span>
      <span><span class="tl-dot is-confirmed tier-medium"></span> medium</span>
      <span><span class="tl-dot is-confirmed tier-low"></span> low</span>
      <span><span class="tl-dot is-tbd"></span> TBD</span>
    </div>
  `;
}

function phaseShort(phase: string): string {
  const map: Record<string, string> = {
    friendly: "FRND",
    group_stage: "GS",
    round_of_32: "R32",
    round_of_16: "R16",
    quarter_final: "QF",
    semi_final: "SF",
    bronze_final: "3rd",
    final: "F",
  };
  return map[phase] ?? phase;
}

function monthDay(iso: string): string {
  const m = iso.match(/^(\d{4})-(\d{2})-(\d{2})/);
  if (!m) return iso;
  const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
  return `${months[parseInt(m[2], 10) - 1]} ${parseInt(m[3], 10)}`;
}
```

```typescript
export function renderConfirmedTile(match: MatchObject): string {
  if (match.teams.confirmed === null || match.teams.confirmed.length !== 2) {

  }
  const [a, b] = match.teams.confirmed;

             id="match-${match.id}" data-signature="${match.signature}">
      <header class="tile-header">
        <span class="tile-meta">

          <strong>${phaseLabel(match.phase)}</strong>

        <span class="tile-demand demand-${tier}">⚑ ${tierLabel}</span>
      </header>
      <div class="confirmed-pair">
        ${teamCol(a)}
        <div class="vs">vs</div>
        ${teamCol(b)}
      </div>
      <footer class="tile-footer">
        <span>${ticketLabel(match.tickets)}</span>
        <span class="tile-venue">${match.venue}</span>

function ticketLabel(t: Tickets): string {
  const parts: string[] = [];
  if (t.suite) parts.push(`${t.suite} suite`);

  if (t.club) parts.push(t.club);
  if (t.split_with) parts.push(`split with ${t.split_with}`);
  return parts.join(" · ") || "—";
}

function phaseLabel(phase: string): string {
```

```typescript
function renderCrossProductRow(s: TbdScenario): string {
  const arrow = arrowFor(s.delta_pp);
  const widthPct = (s.probability * 100).toFixed(1);

      <span class="cp-rank">#${s.rank}</span>
      <img class="flag flag-sm" src="${flagPath(s.team_a.code)}" alt="${flagAltText(s.team_a.code, s.team_a.name)}" />

      <span class="cp-pct">${formatProbability(s.probability)}</span>

    </li>

function arrowFor(delta_pp: number): string {
  if (delta_pp >= 3) return `<span class="delta delta-up" title="+${delta_pp.toFixed(1)}pp">▲</span>`;
  if (delta_pp <= -3) return `<span class="delta delta-down" title="${delta_pp.toFixed(1)}pp">▼</span>`;
  return "";
}

function renderCountdown(days: number | null): string {
  if (days === null) return "";
  let cls = "countdown-neutral";
  if (days <= 1) cls = "countdown-urgent";
  else if (days <= 3) cls = "countdown-warn";
  return `<span class="tile-countdown ${cls}">⏱ Decides in ${days} day${days === 1 ? "" : "s"}</span>`;
}

function slotContextLabel(match: MatchObject): string {
  const fd = match.teams.feeder_distributions;
  if (fd && fd.length === 2) {
    return `${fd[0].label} × ${fd[1].label}`;
  }
  if (fd && fd.length === 1) {
    return fd[0].label;
  }
  return "TBD slot — feeders awaiting bracket resolution";
}

function ticketLabel(m: MatchObject): string {
  const t = m.tickets;
  const parts: string[] = [];
  if (t.suite) parts.push(`${t.suite} suite`);
  if (t.stadium) parts.push(`${t.stadium} stadium`);
  if (t.club) parts.push(t.club);
  if (t.split_with) parts.push(`split with ${t.split_with}`);
  return parts.join(" · ") || "—";
}

function phaseLabel(phase: string): string {
  const map: Record<string, string> = {
    friendly: "Friendly",
    group_stage: "Group Stage",
    round_of_32: "Round of 32",
    round_of_16: "Round of 16",
    quarter_final: "Quarter Final",
    semi_final: "Semi Final",
    bronze_final: "Bronze Final",
    final: "Final",
  };
  return map[phase] ?? phase;
}
```

```typescript
import type { MatchObject } from "../types.js";
import { flagPath, flagAltText } from "../utils/flags.js";
import { formatLocalKickoff } from "../utils/format.js";

export function pickNextUp(matches: MatchObject[], now: Date = new Date()): MatchObject | null {
  if (matches.length === 0) return null;
  const sixHoursAgo = now.getTime() - 6 * 60 * 60 * 1000;
  const candidates = matches.filter(
    (m) => new Date(m.kickoff_utc).getTime() >= sixHoursAgo,
  );
  if (candidates.length === 0) return null;
  candidates.sort(
    (a, b) => new Date(a.kickoff_utc).getTime() - new Date(b.kickoff_utc).getTime(),
  );
  return candidates[0];
}

export function renderHero(match: MatchObject | null): string {
  if (match === null) {
    return `<div class="hero-empty">Tournament complete · no upcoming match</div>`;

      </div>
      ${match.brief ? `<p class="hero-headline">${escape(match.brief.headline)}</p>` : ""}
      <a class="hero-cta" href="#match-${match.id}">View full brief & prep ▸</a>
    </article>
  `;
}

function renderConfirmedHero(match: MatchObject): string {

        <span>${b.name}</span>
      </div>
    </div>
  `;
}

function renderTbdHero(match: MatchObject): string {
  const fd = match.teams.feeder_distributions;
  if (!fd || fd.length === 0) return `<div class="hero-tbd">Knockout slot · TBD</div>`;
  const labels = fd.map((f) => f.label).join(" × ");

    </div>
  `;
}

function escape(s: string): string {
  return s.replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]!));
}
```

```typescript
import type { HostCity, MatchObject, Status } from "../types.js";

export interface FilterState {
  city: HostCity | "all";
  status: Status | "all";
  phase: string | "all";
}

export const DEFAULT_FILTERS: FilterState = { city: "all", status: "all", phase: "all" };

export function renderFilters(state: FilterState): string {
  return `
    <div class="filters">
      <label>City
        <select data-filter="city">
          ${cityOptions(state.city)}
        </select>
      </label>
      <label>Status
        <select data-filter="status">
          <option value="all"${state.status === "all" ? " selected" : ""}>All</option>
          <option value="confirmed"${state.status === "confirmed" ? " selected" : ""}>Confirmed</option>
          <option value="tbd"${state.status === "tbd" ? " selected" : ""}>TBD</option>
        </select>
      </label>
      <label>Phase
        <select data-filter="phase">
          ${phaseOptions(state.phase)}
        </select>
      </label>
    </div>
  `;
}

function cityOptions(selected: string): string {
  const cities = ["all", "Atlanta", "NY/NJ", "Miami"];
  return cities
    .map((c) => `<option value="${c}"${c === selected ? " selected" : ""}>${c === "all" ? "All cities" : c}</option>`)
    .join("");
}

function phaseOptions(selected: string): string {
  const phases = [
    "all",
    "friendly",
    "group_stage",
    "round_of_32",
    "round_of_16",
    "quarter_final",
    "semi_final",
    "bronze_final",
    "final",
  ];
  return phases
    .map(
      (p) =>
        `<option value="${p}"${p === selected ? " selected" : ""}>${p === "all" ? "All phases" : p.replace(/_/g, " ")}</option>`,
    )
    .join("");
}

export function applyFilters(matches: MatchObject[], state: FilterState): MatchObject[] {
  return matches.filter((m) => {
    if (state.city !== "all" && m.host_city !== state.city) return false;
    if (state.status !== "all" && m.status !== state.status) return false;
    if (state.phase !== "all" && m.phase !== state.phase) return false;
    return true;
  });
}
```

```typescript
import { fetchMatches, type FetchResult } from "./data/fetcher.js";
import { changedMatchIds } from "./data/diff.js";
import { renderHeader } from "./components/header.js";
import { renderFooter } from "./components/footer.js";
import { renderTimeline } from "./components/timeline.js";
import { renderConfirmedTile } from "./components/confirmedTile.js";
import { renderTbdTile } from "./components/tbdTile.js";
import { pickNextUp, renderHero } from "./components/hero.js";
import { renderExpandedDetail } from "./components/expandedDetail.js";
import {
  DEFAULT_FILTERS,
  applyFilters,
  renderFilters,
  type FilterState,
} from "./components/filters.js";
import { computeWallClockFreshness } from "./utils/format.js";
import type { MatchObject, MatchesFile } from "./types.js";

const POLL_INTERVAL_MS = 60_000;
const DATA_URL = "data/matches.json";

let lastFile: MatchesFile | null = null;
let filters: FilterState = { ...DEFAULT_FILTERS };
const expandedIds = new Set<string>();

function renderHeaderFooter(file: MatchesFile): void {
  const wallFreshness = computeWallClockFreshness(
    file.generated_at,
    file.data_freshness,
    file.tournament_phase,
  );
  document.getElementById("app-header")!.innerHTML = renderHeader({
    generatedAt: file.generated_at,
    freshness: wallFreshness,
  });
  document.getElementById("app-footer")!.innerHTML = renderFooter({
    generatedAt: file.generated_at,
  });
}

function renderUnreachableHeader(): void {
  document.getElementById("app-header")!.innerHTML = renderHeader({
    generatedAt: new Date().toISOString(),
    freshness: "unreachable",
  });
}

function renderTimelineAndHero(file: MatchesFile): void {
  document.getElementById("timeline")!.innerHTML = renderTimeline(file.matches);
  document.getElementById("hero")!.innerHTML = renderHero(pickNextUp(file.matches));
}

function renderFiltersUI(): void {
  document.getElementById("filters")!.innerHTML = renderFilters(filters);
}

function renderTile(m: MatchObject): string {
  return m.status === "confirmed" ? renderConfirmedTile(m) : renderTbdTile(m);
}

function renderMatchList(file: MatchesFile): void {
  const visible = applyFilters(file.matches, filters);
  const matchList = document.getElementById("match-list")!;
  matchList.innerHTML = visible.map(renderTile).join("");
  // Re-apply expanded state
  for (const id of expandedIds) {
    const tile = document.getElementById(`match-${id}`);
    const match = file.matches.find((m) => m.id === id);
    if (tile && match) {
      tile.insertAdjacentHTML("beforeend", renderExpandedDetail(match));
      tile.classList.add("is-expanded");
    }
  }
}

function diffRenderMatchList(prev: MatchesFile, next: MatchesFile): void {
  const visible = applyFilters(next.matches, filters);
  const visibleIds = new Set(visible.map((m) => m.id));
  const matchList = document.getElementById("match-list")!;

  // Remove tiles for matches that are now filtered out
  for (const tile of Array.from(matchList.querySelectorAll<HTMLElement>(".match-tile"))) {
    const id = tile.id.replace(/^match-/, "");
    if (!visibleIds.has(id)) tile.remove();
  }

  const ids = changedMatchIds(prev, next);
  const nextById = new Map(next.matches.map((m) => [m.id, m] as const));

  for (const id of ids) {
    const match = nextById.get(id);
    if (!match || !visibleIds.has(id)) continue;
    const existing = document.getElementById(`match-${id}`);
    const html = renderTile(match);
    if (existing) {
      existing.outerHTML = html;
    } else {
      // Match wasn't in DOM before (e.g., filter just changed). Append.
      matchList.insertAdjacentHTML("beforeend", html);
    }
    if (expandedIds.has(id)) {
      const tile = document.getElementById(`match-${id}`);
      tile?.insertAdjacentHTML("beforeend", renderExpandedDetail(match));
      tile?.classList.add("is-expanded");
    }
  }
}

function attachInteractions(): void {
  const matchList = document.getElementById("match-list")!;
  matchList.addEventListener("click", (ev) => {
    const target = ev.target as Element;
    if (target.closest("a, details > summary")) return;
    const tile = target.closest<HTMLElement>(".match-tile");
    if (!tile) return;
    const id = tile.id.replace(/^match-/, "");
    const match = lastFile?.matches.find((m) => m.id === id);
    if (!match) return;

    const existing = tile.querySelector(".expanded-detail");
    if (existing) {
      existing.remove();
      tile.classList.remove("is-expanded");
      expandedIds.delete(id);
      if (location.hash === `#match-${id}`) history.replaceState(null, "", " ");
    } else {
      tile.insertAdjacentHTML("beforeend", renderExpandedDetail(match));
      tile.classList.add("is-expanded");
      expandedIds.add(id);
      history.replaceState(null, "", `#match-${id}`);
    }
  });

  document.getElementById("filters")!.addEventListener("change", (ev) => {
    const target = ev.target as HTMLSelectElement;
    const key = target.dataset.filter;
    if (key === "city" || key === "status" || key === "phase") {
      filters = { ...filters, [key]: target.value };
      if (lastFile) renderMatchList(lastFile);
    }
  });
}

function autoExpandFromHash(): void {
  const hashId = location.hash.replace(/^#match-/, "");
  if (!hashId || !lastFile) return;
  const tile = document.getElementById(`match-${hashId}`);
  if (!tile) return;
  tile.scrollIntoView({ behavior: "smooth", block: "start" });
  if (!tile.querySelector(".expanded-detail")) tile.click();
}

async function poll(): Promise<void> {
  const result: FetchResult = await fetchMatches(DATA_URL);
  if (result.kind === "unreachable") {
    renderUnreachableHeader();
    return;
  }
  const file = result.file;
  renderHeaderFooter(file);
  renderTimelineAndHero(file);

  if (lastFile === null) {
    renderMatchList(file);
  } else {
    diffRenderMatchList(lastFile, file);
  }
  lastFile = file;
}

async function bootstrap(): Promise<void> {
  renderFiltersUI();
  attachInteractions();
  await poll();
  autoExpandFromHash();
  setInterval(() => void poll(), POLL_INTERVAL_MS);
}

void bootstrap();
```
